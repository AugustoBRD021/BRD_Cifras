int vInteiro;

vInteiro = entrada.nextInt(); //aqui é onde o inteiro inserido pelo usuário é armazenado na variável

double vDouble;

System.out.printf("Informe um Double: ");

vDouble = entrada.nextDouble(); //aqui é onde o Double inserido pelo usuário é armazenado na variável (o numero com decimal devera ser separado por vírgula, e não ponto)

System.out.print("O double é : " + vDouble);

float vFloat;

System.out.printf("Informe um float: ");

System.out.print("O float é : " + vFloat);


Exemplo de código:


import java.util.Scanner; //importamos a biblioteca para que o usuário possa entrar com dados pelo teclado

public class Main
{
	public static void main(String[] args) 
	{
		int numA, numB; //declaramos as variáveis a serem utilizadas no código
		
		Scanner entr = new Scanner(System.in); //inicializamos o scanner para receber os dados do usuario
		
		System.out.printf("Insira um numero: ");
		numA = entr.nextInt(); //entrada de dados do usuário 
		
		System.out.printf("Insira outro numero: ");
		numB = entr.nextInt(); //entrada de dados do usuário 
		
        System.out.printf("O primeiro numero é %d e o segundo é %d%n", numA, numB); //quando chamamos o "%d", estamos nos referindo a variável
											do tipo "int", e colocamos o format no print (printf),
											utilizamos o "n" em "%dn" para indicar uma quebra de linha (%n),
											e, após tudo, fechamos a linha colocando na ordem correta as
											variáveis a serem impressas na tela.
		
	}
}
